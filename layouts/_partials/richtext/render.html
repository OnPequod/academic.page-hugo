{{- /* Renders an API rich-text value ({html, json}) and returns an HTML string. By default it uses "html", which Academic.page renders and sanitizes the same way the public site does. With params.academicPage.richText = "json", or when "html" is missing, it renders the Tiptap (ProseMirror) document with node.html, which supports a safe subset and escapes all text. */ -}} {{- $out := "" -}} {{- with . -}} {{- $mode := site.Params.academicPage.richText | default "html" -}} {{- if and .html (ne $mode "json") -}} {{- $out = .html -}} {{- else if .json -}} {{- $out = partial "richtext/node.html" .json -}} {{- else -}} {{- $out = .html | default "" -}} {{- end -}} {{- end -}} {{- return $out -}}
